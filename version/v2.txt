        // Changer de page
        function showPage(pageId) {
            document.querySelectorAll('.page-section').forEach(p => p.classList.remove('active'));
            document.querySelectorAll('nav a').forEach(a => a.classList.remove('active'));
            document.getElementById(pageId).classList.add('active');
            
            // Activer le lien de la barre de nav s'il existe
            if(document.getElementById('nav-' + pageId)) {
                document.getElementById('nav-' + pageId).classList.add('active');
            }
            
            document.querySelectorAll('.dropdown-menu').forEach(menu => menu.classList.remove('active'));
            window.scrollTo({ top: 0, behavior: 'smooth' }); // Remonter en haut de page
        }

        // Dropdowns
        function toggleMenu(menuId) {
            const menu = document.getElementById(menuId);
            const isCurrentlyActive = menu.classList.contains('active');
            document.querySelectorAll('.dropdown-menu').forEach(m => m.classList.remove('active'));
            
            if (!isCurrentlyActive) {
                menu.classList.add('active');
            }
            if(menuId === 'menu-notifs') {
                const badge = document.querySelector('.notif-badge');
                if(badge) badge.style.display = 'none';
            }
        }

        // Mode Sombre
        function toggleDarkMode(cb) {
            document.body.classList.toggle('light-mode', !cb.checked);
        }

        document.addEventListener('click', function(event) {
            if (!event.target.closest('.header-actions')) {
                document.querySelectorAll('.dropdown-menu').forEach(m => m.classList.remove('active'));
            }
        });

        // --- GESTION DU CALENDRIER INTELLIGENT ---
        const monthNames = ["Janvier", "Février", "Mars", "Avril", "Mai", "Juin", "Juillet", "Août", "Septembre", "Octobre", "Novembre", "Décembre"];
        let currentMonth = 3; // On démarre en Avril 2026 (index 3)
        let currentYear = 2026;

        function changeMonth(dir) {
            currentMonth += dir;
            if (currentMonth > 11) { currentMonth = 0; currentYear++; }
            if (currentMonth < 0) { currentMonth = 11; currentYear--; }
            renderCalendar();
        }

        // --- GESTION DE LA POP-UP (MODAL) ---
        function openDayModal(day, month, year, eventText) {
            const modal = document.getElementById('calendar-modal');
            const dateText = document.getElementById('modal-date-text');
            const descText = document.getElementById('modal-desc-text');
            
            // On écrit la date en haut
            dateText.innerText = day + " " + monthNames[month] + " " + year;
            
            // Si l'événement est vide, on affiche "Rien ce jour là"
            if (!eventText || eventText === "") {
                descText.innerHTML = "Rien ce jour-là.";
                descText.style.color = "var(--text-dim)";
            } else {
                descText.innerHTML = eventText;
                descText.style.color = "var(--accent-gold)";
            }
            
            modal.classList.add('active');
        }

        // --- GESTION DES ALERTES CUSTOM (Nouveau) ---
        function showAlertModal(title, message) {
            const modal = document.getElementById('calendar-modal');
            const dateText = document.getElementById('modal-date-text');
            const descText = document.getElementById('modal-desc-text');
            
            // On utilise la même fenêtre mais on change le texte et la couleur
            dateText.innerText = title;
            descText.innerHTML = message;
            descText.style.color = "var(--text-main)"; // Couleur normale pour les alertes
            
            modal.classList.add('active');
        }

        function closeModal(event) {
            // Fermer si on clique sur le fond sombre
            if(event.target.id === 'calendar-modal') {
                document.getElementById('calendar-modal').classList.remove('active');
            }
        }
        
        function forceCloseModal() {
            // Fermer si on clique sur la croix
            document.getElementById('calendar-modal').classList.remove('active');
        }

        // --- GÉNÉRATION DE LA GRILLE DU CALENDRIER ---
        function renderCalendar() {
            document.getElementById("cal-title").innerText = monthNames[currentMonth] + " " + currentYear;
            const grid = document.getElementById("cal-days");
            grid.innerHTML = "";

            // Noms des jours
            const days = ["Lun", "Mar", "Mer", "Jeu", "Ven", "Sam", "Dim"];
            days.forEach(d => {
                grid.innerHTML += `<div class="cal-day-name">${d}</div>`;
            });

            // Calcul du 1er jour du mois
            let firstDay = new Date(currentYear, currentMonth, 1).getDay();
            firstDay = firstDay === 0 ? 6 : firstDay - 1; 
            
            const daysInMonth = new Date(currentYear, currentMonth + 1, 0).getDate();

            // Cases vides
            for(let i = 0; i < firstDay; i++) {
                grid.innerHTML += `<div class="cal-case" style="visibility:hidden"></div>`;
            }

            // Remplissage des jours
            for(let i = 1; i <= daysInMonth; i++) {
                let extraHtml = "";
                let classList = "cal-case";
                let eventForModal = "";
                
                // 18 Avril 2026 (Aujourd'hui)
                if(currentMonth === 3 && currentYear === 2026 && i === 18) {
                    classList += " today";
                    extraHtml = `<div style="font-size: 0.7em; color: var(--accent-gold); font-weight: bold; text-align: center; margin-top: auto;">Aujourd'hui</div>`;
                    eventForModal = "Aujourd'hui";
                }
                
                // 18 Février 2026 (Début Plopsa)
                if(currentMonth === 1 && currentYear === 2026 && i === 18) {
                    extraHtml = `<div class="cal-event pink">Début Plopsa</div>`;
                    eventForModal = "Début du chantier Plopsaland ! 🎢";
                }
                
                // 11 Avril 2026 (Réunion Staff)
                if(currentMonth === 3 && currentYear === 2026 && i === 11) {
                    extraHtml = `<div class="cal-event">Réunion Staff</div>`;
                    eventForModal = "Réunion Staff ExoWorld 🛠️";
                }

                // La magie : on ajoute onclick pour ouvrir la popup
                grid.innerHTML += `
                    <div class="${classList}" onclick="openDayModal(${i}, ${currentMonth}, ${currentYear}, '${eventForModal}')">
                        <div class="cal-date" ${classList.includes('today') ? 'style="color: var(--accent-gold);"' : ''}>${i}</div>
                        ${extraHtml}
                    </div>
                `;
            }
        }

        // Lancer le calendrier au chargement de la page
        window.onload = renderCalendar;

    </script>
