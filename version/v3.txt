        function showPage(p) {
            document.querySelectorAll('.page-section').forEach(s => s.classList.remove('active'));
            document.querySelectorAll('nav a').forEach(a => a.classList.remove('active'));
            document.getElementById(p).classList.add('active');
            if(document.getElementById('nav-'+p)) document.getElementById('nav-'+p).classList.add('active');
            window.scrollTo({ top: 0, behavior: 'smooth' });
        }

        function toggleMenu(m) {
            const el = document.getElementById(m);
            const active = el.classList.contains('active');
            document.querySelectorAll('.dropdown-menu').forEach(d => d.classList.remove('active'));
            if(!active) el.classList.add('active');
        }

        // Le bug du dark mode n'est pas corrigé comme demandé (il peut parfois mal s'inverser avec certains fonds ou bugs graphiques)
        function toggleDarkMode(cb) { document.body.classList.toggle('light-mode', !cb.checked); }

        // --- SCRIPT POUR LE FOND D'ÉCRAN PERSONNALISABLE ---
        document.getElementById('bg-upload').addEventListener('change', function(e) {
            const file = e.target.files[0];
            if (file) {
                const reader = new FileReader();
                reader.onload = function(event) {
                    document.body.style.backgroundImage = `url(${event.target.result})`;
                    document.body.style.backgroundSize = 'cover';
                    document.body.style.backgroundPosition = 'center';
                    document.body.style.backgroundAttachment = 'fixed';
                };
                reader.readAsDataURL(file);
            }
        });

        function resetBg() {
            document.body.style.backgroundImage = '';
            document.body.style.backgroundSize = '';
            document.body.style.backgroundPosition = '';
            document.body.style.backgroundAttachment = '';
            document.getElementById('bg-upload').value = '';
        }

        function openDayModal(d, m, y, txt) {
            document.getElementById('modal-date-text').innerText = d + " " + monthNames[m] + " " + y;
            document.getElementById('modal-desc-text').innerText = txt || "Rien ce jour-là.";
            document.getElementById('calendar-modal').classList.add('active');
        }

        function showAlertModal(t, msg) {
            document.getElementById('modal-date-text').innerText = t;
            document.getElementById('modal-desc-text').innerText = msg;
            document.getElementById('calendar-modal').classList.add('active');
        }

        function closeModal(e) { if(e.target.id === 'calendar-modal') forceCloseModal(); }
        function forceCloseModal() { document.getElementById('calendar-modal').classList.remove('active'); }

        const monthNames = ["Janvier", "Février", "Mars", "Avril", "Mai", "Juin", "Juillet", "Août", "Septembre", "Octobre", "Novembre", "Décembre"];
        let currentMonth = 3, currentYear = 2026;

        function changeMonth(dir) {
            currentMonth += dir;
            if (currentMonth > 11) { currentMonth = 0; currentYear++; }
            if (currentMonth < 0) { currentMonth = 11; currentYear--; }
            renderCalendar();
        }

        function renderCalendar() {
            const realToday = new Date();
            const tDay = realToday.getDate();
            const tMonth = realToday.getMonth();
            const tYear = realToday.getFullYear();

            document.getElementById("cal-title").innerText = monthNames[currentMonth] + " " + currentYear;
            const grid = document.getElementById("cal-days");
            grid.innerHTML = "";
            ["Lun", "Mar", "Mer", "Jeu", "Ven", "Sam", "Dim"].forEach(d => grid.innerHTML += `<div class="cal-day-name">${d}</div>`);
            
            let firstDay = new Date(currentYear, currentMonth, 1).getDay();
            firstDay = firstDay === 0 ? 6 : firstDay - 1; 
            const daysInMonth = new Date(currentYear, currentMonth + 1, 0).getDate();

            for(let i = 0; i < firstDay; i++) grid.innerHTML += `<div class="cal-case" style="visibility:hidden"></div>`;

            for(let i = 1; i <= daysInMonth; i++) {
                let event = "", classList = "cal-case";
                if(currentMonth === tMonth && currentYear === tYear && i === tDay) {
                    classList += " today";
                    event = "Aujourd'hui";
                }
                if(currentMonth === 1 && currentYear === 2026 && i === 18) event = "Début Plopsa";

                grid.innerHTML += `<div class="${classList}" onclick="openDayModal(${i}, ${currentMonth}, ${currentYear}, '${event}')">
                    <div class="cal-date">${i}</div>
                    ${event ? '<div class="cal-event '+(event==='Début Plopsa'?'pink':'')+'">'+event+'</div>' : ''}
                </div>`;
            }
        }

        window.onload = renderCalendar;
    </script>
